    "## Customer Segmentation\n",
    "Using K-Means clustering to segment customers"
   ]
  },
  {
   "cell_type": "code",
   "execution_count": 5,
   "metadata": {},
   "source": [
    "# Aggregate customer metrics\n",
    "customer_metrics = df.groupby('customer_id').agg({\n",
    "    'transaction_amount': ['sum', 'mean', 'count'],\n",
    "    'transaction_date': lambda x: (x.max() - x.min()).days\n",
    "}).reset_index()\n",
    "\n",
    "customer_metrics.columns = ['customer_id', 'total_spend', 'avg_spend', \n",
    "                             'transaction_count', 'customer_lifetime_days']\n",
    "\n",
    "# Standardize features\n",
    "scaler = StandardScaler()\n",
    "features = ['total_spend', 'avg_spend', 'transaction_count', 'customer_lifetime_days']\n",
    "X_scaled = scaler.fit_transform(customer_metrics[features])\n",
    "\n",
    "# K-Means clustering\n",
    "kmeans = KMeans(n_clusters=4, random_state=42)\n",
    "customer_metrics['cluster'] = kmeans.fit_predict(X_scaled)\n",
    "\n",
    "print('Customer segmentation complete')\n",
    "print('\\nCluster sizes:')\n",
    "print(customer_metrics['cluster'].value_counts())"
   ],
