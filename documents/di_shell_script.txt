# Application settings
APP_NAME="enterprise-portal"
APP_VERSION="2.1.0"
DEPLOY_ENV="production"
DEPLOY_DATE=$(date +%Y%m%d_%H%M%S)

# Server settings
SERVER_HOST="app.company.com"
SERVER_PORT="8080"
DEPLOY_USER="deploy"

# Directory paths
APP_DIR="/opt/applications/${APP_NAME}"
LOG_DIR="/var/log/${APP_NAME}"
BACKUP_DIR="/backup/${APP_NAME}"

# Database configuration
DB_HOST="db.internal.company.com"
DB_PORT="5432"
DB_NAME="production_db"
DB_USER="app_user"
# DB_PASSWORD will be loaded from environment

# Load environment variables from secure vault
if [ -f "/etc/secrets/app.env" ]; then
    source /etc/secrets/app.env
    echo "Environment variables loaded successfully"
else
    echo "ERROR: Environment file not found"
    exit 1
fi

# ==============================================
# Functions
# ==============================================

# Function: Create backup
create_backup() {
    echo "Creating backup..."
    BACKUP_FILE="${BACKUP_DIR}/backup_${DEPLOY_DATE}.tar.gz"
    
    if [ -d "${APP_DIR}" ]; then
        tar -czf "${BACKUP_FILE}" -C "${APP_DIR}" .
        echo "Backup created: ${BACKUP_FILE}"
    else
        echo "No existing deployment found, skipping backup"
    fi
}

# Function: Stop application
stop_application() {
    echo "Stopping application..."
    systemctl stop ${APP_NAME} || true
    sleep 2
    echo "Application stopped"
}

# Function: Deploy new version
deploy_application() {
    echo "Deploying version ${APP_VERSION}..."
    
    # Create directory if not exists
    mkdir -p "${APP_DIR}"
    mkdir -p "${LOG_DIR}"
    
    # Copy application files
    cp -r ./build/* "${APP_DIR}/"
    
    # Set permissions
    chown -R ${DEPLOY_USER}:${DEPLOY_USER} "${APP_DIR}"
    chmod 755 "${APP_DIR}"
    
    echo "Application deployed successfully"
}

# Function: Update configuration
update_configuration() {
    echo "Updating configuration..."
    
    cat > "${APP_DIR}/config.yml" <<EOF
application:
  name: ${APP_NAME}
  version: ${APP_VERSION}
  environment: ${DEPLOY_ENV}

server:
  host: ${SERVER_HOST}
  port: ${SERVER_PORT}

database:
  host: ${DB_HOST}
  port: ${DB_PORT}
  name: ${DB_NAME}
  user: ${DB_USER}
  password: \${DB_PASSWORD}

logging:
  level: info
  directory: ${LOG_DIR}
EOF
    
    echo "Configuration updated"
}

# Function: Run database migrations
run_migrations() {
    echo "Running database migrations..."
    
    # Connect to database and run migrations
    PGPASSWORD="${DB_PASSWORD}" psql -h "${DB_HOST}" \
        -p "${DB_PORT}" \
        -U "${DB_USER}" \
        -d "${DB_NAME}" \
        -f "./migrations/latest.sql"
    
    echo "Migrations completed"
}

# Function: Start application
start_application() {
    echo "Starting application..."
    systemctl start ${APP_NAME}
    sleep 3
    
    # Verify application is running
    if systemctl is-active --quiet ${APP_NAME}; then
        echo "Application started successfully"
    else
        echo "ERROR: Application failed to start"
        exit 1
    fi
}

# Function: Health check
health_check() {
    echo "Performing health check..."
    
    MAX_ATTEMPTS=10
    ATTEMPT=0
    
    while [ $ATTEMPT -lt $MAX_ATTEMPTS ]; do
        if curl -f -s "http://localhost:${SERVER_PORT}/health" > /dev/null; then
            echo "Health check passed"
            return 0
        fi
        
        ATTEMPT=$((ATTEMPT + 1))
        echo "Attempt ${ATTEMPT}/${MAX_ATTEMPTS} failed, retrying..."
        sleep 5
    done
    
    echo "ERROR: Health check failed after ${MAX_ATTEMPTS} attempts"
    return 1
}

# ==============================================
# Main Deployment Process
# ==============================================

main() {
    echo "=========================================="
    echo "Starting deployment: ${APP_NAME} v${APP_VERSION}"
    echo "Environment: ${DEPLOY_ENV}"
    echo "Date: ${DEPLOY_DATE}"
    echo "=========================================="
    
    create_backup
    stop_application
    deploy_application
    update_configuration
    run_migrations
    start_application
    
    if health_check; then
        echo "=========================================="
        echo "Deployment completed successfully!"
        echo "=========================================="
        exit 0
    else
        echo "=========================================="
        echo "Deployment failed! Rolling back..."
        echo "=========================================="
        # Rollback logic would go here
        exit 1
    fi
}

# Execute main function
main "$@"
