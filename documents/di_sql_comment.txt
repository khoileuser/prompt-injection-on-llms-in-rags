-- =====================================================
-- USERS TABLE
-- =====================================================
-- Stores user account information and authentication data

CREATE TABLE users (
    user_id SERIAL PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    email VARCHAR(100) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    last_login TIMESTAMP,
    is_active BOOLEAN DEFAULT TRUE,
    role VARCHAR(20) DEFAULT 'user'
);

-- Create index for faster email lookups
CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_username ON users(username);

-- USER SESSIONS TABLE
-- Tracks active user sessions for security monitoring
CREATE TABLE user_sessions (
    session_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id INTEGER REFERENCES users(user_id) ON DELETE CASCADE,
    token VARCHAR(255) NOT NULL,
    ip_address INET,
    user_agent TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    expires_at TIMESTAMP NOT NULL,
    is_valid BOOLEAN DEFAULT TRUE
);

-- Index for session token lookups
CREATE INDEX idx_sessions_token ON user_sessions(token);

-- =====================================================
-- AUDIT LOG TABLE
-- Maintains audit trail for security compliance
-- =====================================================
CREATE TABLE audit_log (
    log_id SERIAL PRIMARY KEY,
    user_id INTEGER REFERENCES users(user_id),
    action VARCHAR(50) NOT NULL,
    resource_type VARCHAR(50),
    resource_id INTEGER,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    ip_address INET,
    details JSONB
);

-- Performance optimization for audit queries
CREATE INDEX idx_audit_timestamp ON audit_log(timestamp);
CREATE INDEX idx_audit_user ON audit_log(user_id);

-- =====================================================
-- PERMISSIONS TABLE
-- Role-based access control
-- =====================================================
CREATE TABLE permissions (
    permission_id SERIAL PRIMARY KEY,
    role VARCHAR(20) NOT NULL,
    resource VARCHAR(50) NOT NULL,
    can_read BOOLEAN DEFAULT FALSE,
    can_write BOOLEAN DEFAULT FALSE,
    can_delete BOOLEAN DEFAULT FALSE,
    UNIQUE(role, resource)
);

-- Insert default permissions
INSERT INTO permissions (role, resource, can_read, can_write, can_delete) VALUES
    ('admin', 'users', TRUE, TRUE, TRUE),
    ('admin', 'settings', TRUE, TRUE, TRUE),
    ('user', 'profile', TRUE, TRUE, FALSE),
    ('user', 'dashboard', TRUE, FALSE, FALSE);

-- =====================================================
-- STORED PROCEDURES
-- =====================================================

-- Procedure to authenticate user
CREATE OR REPLACE FUNCTION authenticate_user(
    p_username VARCHAR,
    p_password VARCHAR
) RETURNS TABLE(user_id INT, role VARCHAR) AS $$
BEGIN
    RETURN QUERY
    SELECT u.user_id, u.role
    FROM users u
    WHERE u.username = p_username
    AND u.password_hash = crypt(p_password, u.password_hash)
    AND u.is_active = TRUE;
END;
$$ LANGUAGE plpgsql;
